========================================
b) Creating an External Hive Table
========================================

Step 8: Create External Table

CREATE TABLE flight(
    fno INT,
    year INT,
    des varchar(20),
    delay FLOAT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
LINES TERMINATED BY '\n'
STORED AS TEXTFILE;

========================================
JOIN OPERATIONS
========================================

1. INNER JOIN

SELECT f.fno, f.year, f1.src, f.des, f.delay
FROM flight f
INNER JOIN flight1 f1
ON f.fno = f1.fno;

----------------------------------------

2. LEFT OUTER JOIN

SELECT f.fno, f.year, f1.src, f.des
FROM flight f
LEFT OUTER JOIN flight1 f1
ON f.fno = f1.fno;

----------------------------------------

3. RIGHT OUTER JOIN

SELECT f.fno, f.year, f1.src, f.des
FROM flight f
RIGHT OUTER JOIN flight1 f1
ON f.fno = f1.fno;

----------------------------------------

4. FULL OUTER JOIN

SELECT f.fno, f.year, f1.src, f.des
FROM flight f
FULL OUTER JOIN flight1 f1
ON f.fno = f1.fno;

----------------------------------------

5. LEFT SEMI JOIN

SELECT *
FROM flight f
LEFT SEMI JOIN flight1 f1
ON f.fno = f1.fno;

----------------------------------------

6. CROSS JOIN

SELECT *
FROM flight
CROSS JOIN flight1;

========================================
d) Create Index on Flight Information Table
========================================

Step 15: Create Index

CREATE INDEX flight_index
ON TABLE flight(fno)
AS 'org.apache.hadoop.hive.ql.index.compact.CompactIndexHandler'
WITH DEFERRED REBUILD;

----------------------------------------

Step 16: Rebuild Index

ALTER INDEX flight_index
ON flight

----------------------------------------

Step 17: Show Indexes

========================================
e) Find Average Departure Delay in 2005
========================================

Step 18: Average Delay

SELECT AVG(delay) AS average_delay
FROM flight
WHERE year = 2005;

========================================
UPDATE OPERATIONS IN HIVE
========================================

Direct UPDATE (Works only on transactional tables)

UPDATE flight
SET delay = 50
WHERE fno = 101;

----------------------------------------

Alternative Method using INSERT OVERWRITE

INSERT OVERWRITE TABLE flight
SELECT
    fno,
    year,
    des,
    CASE
        WHEN fno = 101 THEN 50.0
        ELSE delay
    END
FROM flight;

----------------------------------------

If airline column exists:

INSERT OVERWRITE TABLE flight
SELECT
    fno,
    year,
    des,
    CASE
        WHEN fno = 101 THEN 50.0
        ELSE delay
    END,
    airline
FROM flight;

========================================
IMPORTANT VIVA POINTS
========================================

1. INNER JOIN → matching rows only

2. LEFT JOIN → all rows from left table

3. RIGHT JOIN → all rows from right table

4. FULL OUTER JOIN → all rows from both tables

5. CROSS JOIN → Cartesian product

6. Hive UPDATE works only on transactional tables

7. External table data is not deleted when table is dropped

8. Managed table data is deleted when table is dropped
